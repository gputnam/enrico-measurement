cmake_minimum_required(VERSION 3.20)
project(makedf CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
find_package(ROOT REQUIRED COMPONENTS Core RIO Tree)
add_executable(makedf src/makedf.cxx src/Columns.cxx src/Derived.cxx src/FSIWeight.cxx)
target_link_libraries(makedf PRIVATE ROOT::Core ROOT::RIO ROOT::Tree)
target_compile_options(makedf PRIVATE -Wall -Wextra -O2)
install(TARGETS makedf DESTINATION bin)
